\begin{frame}{Reinforcement Learning Definition}
  \begin{columns}
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item No fixed input-output pairs to learn from
        \item Agent interacts with environment
        \item Receives reward to indicate if its actions are good or bad
        \item Learn a policy, that maximizes cumulative reward
      \end{itemize}

\subsection{Reinforcement Learning Environment}
\begin{frame}{Reinforcement Learning Environment}
  An environment is modeled as a \textbf{Markov Decision Process (MDP)}

  \begin{itemize}
    \item discrete time steps $t = 0, 1, 2, 3, ...$
    \item At each time step
      \begin{itemize}
        \item environment state $S_t$
        \item reward $R_t$ (if $t \neq 0$)
        \item agent selects action $A_t$
      \end{itemize}
    \item resulting environment in next step is defined by
      \begin{itemize}
        \item probability distribution $p(s'|s,a)$
        \item reward function $r(s,a)$
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{CartPole Environment\footnote[frame]{Gymnasium: \url{https://gymnasium.farama.org/environments/classic_control/cart_pole/}}}
  \begin{columns}
    \begin{column}{0.35\textwidth}

    \begin{column}{0.65\textwidth}
      A pole is attached to a cart moving along a frictionless track.
      \vfill
      \textbf{Goal:} Keep the pole balanced by applying left or right
      forces.
      \vfill
      \textbf{Action space:} $0$ (left), $1$ (right)
      \vfill
      \textbf{Observation space:}
      \begin{itemize}
        \item Cart position $\in [-4.8, 4.8]$
        \item Cart velocity $\in (-\infty, \infty)$
        \item Pole angle $\in [-24^\circ, 24^\circ]$
        \item Pole angular velocity $\in (-\infty, \infty)$
      \end{itemize}
      \vfill
      \textbf{Reward:} $1$ per step while not terminated
      \vfill
      \textbf{Termination:}
      \begin{itemize}
        \item Pole angle exceeds $\pm 12^\circ$
        \item Cart leaves area $[-2.4,2.4]$
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Proximal Policy Optimization}

\subsection{Overview}
\begin{frame}{Overview\footnote[frame]{Schulman et al., \emph{Proximal Policy Optimization Algorithms}, arXiv:1707.06347v2 (2017).}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{PPO is an on-policy actor--critic method}
      \vspace{0.5em}
      \begin{itemize}
        \item Alternate between \textbf{sampling} with the current policy and
              \textbf{optimizing} a surrogate objective
        \item \textbf{Actor} $\pi_\theta(a \mid s)$ chooses actions and is the
              policy we improve
        \item \textbf{Critic} $V_\phi(s)$ estimates state value, used to compute
              advantages $\hat A_t$
        \item Reuse each rollout for several mini-batch epochs, but keep the
              new policy close to the old one
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[
          scale=0.82,
          transform shape,
          box/.style={rectangle, draw=tuwblue!80, thick, rounded corners,
                      minimum height=0.75cm,
                      align=center, font=\small},
          smallbox/.style={rectangle, draw=tuwblue!60, thick, rounded corners,
                           minimum height=0.65cm,
                           align=center, font=\scriptsize},
          leftbox/.style={box, text width=2.75cm},
          rightbox/.style={box, text width=3.05cm},
          leftsmall/.style={smallbox, text width=2.75cm},
          rightsmall/.style={smallbox, text width=3.05cm},
          arrow/.style={-{Latex[length=2mm]}, thick, tuwblue!70},
          node distance=0.55cm and 1.35cm
        ]
        \node[leftbox] (actor) {Actor\\$\pi_{\theta_{\mathrm{old}}}(a\mid s)$};
        \node[rightbox, right=of actor] (env) {Environment\\CartPole};
        \node[rightsmall, below=of env] (rollout) {rollout data\\$s_t,a_t,r_t$\\$\log\pi_{\theta_{\mathrm{old}}}$};
        \node[leftbox, below=of actor] (critic) {Critic\\$V_\phi(s)$};
        \node[rightsmall, below=of rollout] (adv) {advantage\\estimates $\hat A_t$};
        \node[leftsmall] (update) at (critic |- adv) {PPO update\\clipped surrogate};

        \draw[arrow] (actor.east) -- node[midway, above=2pt, font=\scriptsize] {actions} (env.west);
        \draw[arrow] (env.south) -- node[right, font=\scriptsize] {samples} (rollout.north);
        \draw[arrow] (rollout.south) -- (adv.north);
        \draw[arrow] (critic.east) -- node[midway, above=3pt, xshift=0.12cm, font=\scriptsize] {$V(s_t)$} (adv.west);
        \draw[arrow] (adv.west) -- (update.east);
        \draw[arrow] (update.north) -- node[left, font=\scriptsize] {fit $V$} (critic.south);
        \draw[arrow, rounded corners] (update.west) -- ++(-0.45,0) |- node[left, font=\scriptsize] {$\theta\leftarrow$} (actor.west);
      \end{tikzpicture}
      \vspace{0.4em}

      \small
      ``Proximal'' means the update is restricted: PPO removes incentive for
      probability ratios outside $[1-\epsilon,\,1+\epsilon]$.
    \end{column}
  \end{columns}
\end{frame}

\subsection{Evaluation Metrics}
\begin{frame}{Evaluation Metrics}
  \begin{itemize}
    \item \textbf{Mean episode return} --- average reward collected per episode; maximum is $500$ per default
    \item \textbf{Standard Deviation} --- standard deviation across multiple seeds, to judge stability across seeds
    \item \textbf{Success rate} --- episode is defined as "solved", if return $\geq 475$

    \item \textbf{Final evaluation} --- $60$ episodes after training; report above metrics
  \end{itemize}
\end{frame}
