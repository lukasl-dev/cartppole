\subsection{Reinforcement Learning Environment}
\begin{frame}{Reinforcement Learning Environment}
  An environment is modeled as a \textbf{Markov Decision Process (MDP)}

  \begin{itemize}
    \item discrete time steps $t = 0, 1, 2, 3, ...$
    \item At each time step
      \begin{itemize}
        \item environment state $S_t$
        \item reward $R_t$ (if $t \neq 0$)
        \item agent selects action $A_t$
      \end{itemize}
    \item resulting environment in next step is defined by
      \begin{itemize}
        \item probability distribution $p(s'|s,a)$
        \item reward function $r(s,a)$
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Proximal Policy Optimization}

\subsection{Overview}
\begin{frame}{Overview\footnote[frame]{Schulman et al., \emph{Proximal Policy Optimization Algorithms}, arXiv:1707.06347v2 (2017).}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{PPO is an on-policy actor--critic method}
      \vspace{0.5em}
      \begin{itemize}
        \item Alternate between \textbf{sampling} with the current policy and
              \textbf{optimizing} a surrogate objective
        \item \textbf{Actor} $\pi_\theta(a \mid s)$ chooses actions and is the
              policy we improve
        \item \textbf{Critic} $V_\phi(s)$ estimates state value, used to compute
              advantages $\hat{A}_t$
        \item Reuse each rollout for several mini-batch epochs, but keep the
              new policy close to the old one
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering

\subsection{Objective}
\begin{frame}{Objective: Policy Loss}
  The actor is updated with the clipped surrogate objective from PPO.

  \vspace{0.6em}
  \begin{block}{Clipped policy objective}
    \begin{align*}
      L_t^{\mathrm{CLIP}}(\theta)
      &= \min\!\left(
          r_t(\theta)\hat{A}_t,
          \operatorname{clip}\bigl(r_t(\theta), 1-\epsilon, 1+\epsilon\bigr)\hat{A}_t
        \right) \\
      r_t(\theta)
      &= \frac{\pi_\theta(a_t\mid s_t)}{\pi_{\theta_{\mathrm{old}}}(a_t\mid s_t)}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $r_t(\theta)$ compares the new policy to the policy that collected the rollout
    \item $\hat{A}_t$ tells whether the sampled action was better or worse than expected
    \item Clipping removes incentive for overly large policy changes
  \end{itemize}
\end{frame}

\begin{frame}{Objective: Value Loss}
  The critic is trained as a regression problem: predict the return target for
  each sampled state.

  \vspace{0.6em}
  \begin{block}{Value-function loss}
    \begin{align*}
      L_t^{\mathrm{VF}}(\phi)
      &= \left(V_\phi(s_t) - V_t^{\mathrm{target}}\right)^2
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $V_\phi(s_t)$ estimates how much return is expected from state $s_t$
    \item $V_t^{\mathrm{target}}$ is computed from rollout rewards and bootstrapping
    \item A better critic gives better advantage estimates and reduces update variance
  \end{itemize}
\end{frame}

\begin{frame}{Objective: Entropy Bonus}
  PPO also rewards stochastic policies to avoid collapsing to one action too
  early.

  \vspace{0.6em}
  \begin{block}{Entropy bonus}
    \begin{align*}
      S[\pi_\theta](s_t)
      &= -\sum_a \pi_\theta(a\mid s_t)\log \pi_\theta(a\mid s_t)
    \end{align*}
  \end{block}

  \vspace{0.4em}
  \begin{block}{Combined objective}
    \vspace{-0.4em}
    \[
      L_t^{\mathrm{PPO}}(\theta, \phi)
      = L_t^{\mathrm{CLIP}}(\theta)
        - c_1 L_t^{\mathrm{VF}}(\phi)
        + c_2 S[\pi_\theta](s_t)
    \]
    \vspace{-0.8em}
  \end{block}
\end{frame}


\subsection{Advantage Estimation}
\begin{frame}{Advantage Estimation: Why Advantages?}
  The policy loss needs a signal for whether the sampled action was good.
  Instead of using raw return, PPO uses an \textbf{advantage estimate}:

  \vspace{0.4em}
  \begin{block}{General idea}
    \[
      \hat{A}_t \approx
      \underbrace{\widehat{Q}_t}_{\text{rollout estimate}}
      -
      \underbrace{V_\phi(s_t)}_{\text{critic baseline}}
    \]
  \end{block}

  \begin{itemize}
    \item $\widehat{Q}_t$: estimated from rewards after taking $a_t$ in the rollout
    \item $V_\phi(s_t)$: critic's baseline expectation for state $s_t$
    \item $\hat{A}_t > 0$: action was better than expected $\Rightarrow$ make it more likely
    \item $\hat{A}_t < 0$: action was worse than expected $\Rightarrow$ make it less likely
  \end{itemize}
\end{frame}

\begin{frame}{Advantage Estimation: Rollout Return}
  A direct estimate compares a discounted rollout return with the critic's
  current value estimate.

  \vspace{0.5em}
  \begin{block}{Truncated Monte-Carlo-style estimate}
    \begin{align*}
      G_t &= \sum_{l=0}^{T-t-1} \gamma^l r_{t+l}
             + \gamma^{T-t} V_\phi(s_T) \\
      \hat{A}_t &= G_t - V_\phi(s_t)
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $t$ is the current position in a rollout segment of length $T$
    \item $l=0,\ldots,T-t-1$ iterates over future rewards
          $r_t, r_{t+1}, \ldots, r_{T-1}$
    \item Uses real rewards from the sampled trajectory segment
    \item Bootstraps with $V_\phi(s_T)$ because PPO uses fixed-length rollouts
    \item Often low bias, but can have high variance
  \end{itemize}
\end{frame}

\begin{frame}{Advantage Estimation: Bootstrapping}
  \small
  A rollout can stop because we reached the fixed segment length, even if the
  episode is not finished. Then the future after $s_T$ is unobserved.

  \vspace{0.5em}
  \centering
  \begin{tikzpicture}[
      state/.style={circle, draw=tuwblue!80, thick, minimum size=0.72cm, font=\scriptsize},
      reward/.style={rectangle, draw=tuwblue!60, rounded corners, minimum width=0.65cm,
                     minimum height=0.45cm, font=\scriptsize},
      arrow/.style={-{Latex[length=1.6mm]}, thick, tuwblue!70},
      node distance=0.45cm
    ]
    \node[state] (st) {$s_t$};
    \node[reward, right=of st] (rt) {$r_t$};
    \node[state, right=of rt] (st1) {$s_{t+1}$};
    \node[right=0.25cm of st1] (dots) {$\cdots$};
    \node[reward, right=0.25cm of dots] (rT) {$r_{T-1}$};
    \node[state, right=of rT] (sT) {$s_T$};
    \node[reward, right=0.85cm of sT, dashed, draw=failred!70] (future) {$?$};
    \node[below=0.25cm of future, font=\scriptsize, text=failred] {unobserved future};
    \node[reward, below=0.72cm of sT, draw=successgreen!80, text width=1.35cm,
          align=center] (boot) {$V_\phi(s_T)$};

    \draw[arrow] (st) -- (rt);
    \draw[arrow] (rt) -- (st1);
    \draw[arrow] (st1) -- (dots);
    \draw[arrow] (dots) -- (rT);
    \draw[arrow] (rT) -- (sT);
    \draw[arrow, dashed, failred!70] (sT) -- (future);
    \draw[arrow, successgreen!80] (sT) -- (boot);
  \end{tikzpicture}

  \vspace{0.4em}
  \begin{block}{Replace the unknown tail by the critic's estimate}
    \[
      G_t = \underbrace{r_t + \gamma r_{t+1} + \cdots + \gamma^{T-t-1} r_{T-1}}_{\text{observed rewards}}
            + \gamma^{T-t}\underbrace{V_\phi(s_T)}_{\text{estimated future}}
    \]
  \end{block}

  \vspace{-0.2em}
  \begin{itemize}
    \item This replacement is called \textbf{bootstrapping}
    \item If $s_T$ is terminal, the estimated future is $0$
  \end{itemize}
\end{frame}

\begin{frame}{Advantage Estimation: GAE}
  Generalized Advantage Estimation (GAE) smooths many temporal-difference errors
  into one advantage estimate.

  \vspace{0.5em}
  \begin{block}{GAE estimate}
    \begin{align*}
      \delta_t &= r_t + \gamma V(s_{t+1}) - V(s_t) \\
      \hat{A}_t^{\mathrm{GAE}}
      &= \sum_{l=0}^{T-t-1} (\gamma\lambda)^l \delta_{t+l}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item The sum again iterates over future offsets
          $l=0,\ldots,T-t-1$ inside the rollout
    \item $\delta_t$ measures one-step surprise relative to the critic
    \item $\lambda$ controls how far future TD errors influence $\hat{A}_t$
    \item Lower $\lambda$: lower variance; higher $\lambda$: lower bias
  \end{itemize}

  \vspace{0.2em}
  \centering
  \footnotesize Critic target used later: $V_t^{\mathrm{target}}=\hat{A}_t+V(s_t)$.
\end{frame}

\begin{frame}{Advantage Estimation: What Does $\lambda$ Do?}
  In GAE, $\lambda$ controls how much we trust short-term critic estimates
  versus longer rollout returns.

  \vspace{0.5em}
  \begin{block}{Future TD errors are exponentially weighted}
    \[
      \hat{A}_t^{\mathrm{GAE}}
      = \delta_t
        + (\gamma\lambda)\delta_{t+1}
        + (\gamma\lambda)^2\delta_{t+2}
        + \cdots
    \]
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Small $\lambda$}\\[0.2em]
      \begin{itemize}
        \item mostly one-step TD errors
        \item relies more on critic predictions
        \item lower variance, more bias
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Large $\lambda$}\\[0.2em]
      \begin{itemize}
        \item includes more future rewards
        \item closer to Monte-Carlo return
        \item lower bias, more variance
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3em}
  \centering
  \small $\lambda=0$: one-step TD estimate \qquad $\lambda=1$: rollout-return-like estimate
\end{frame}

\subsection{Clipped vs.\ Unclipped Objective}
\begin{frame}{Clipped vs.\ Unclipped Objective}
  \small
  PPO reuses the same rollout for several gradient steps. The ratio measures how
  much the new policy changed on the sampled action:
  \[
    r_t(\theta)=\frac{\pi_\theta(a_t\mid s_t)}{\pi_{\theta_{\mathrm{old}}}(a_t\mid s_t)}
    \qquad \text{and initially } r_t=1.
  \]

  \vspace{-0.3em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Unclipped surrogate}
        \begin{minipage}[c][1.55cm][c]{\textwidth}
          \[
            L_t^{\mathrm{unclip}}(\theta)=r_t(\theta)\hat{A}_t
          \]
        \end{minipage}
      \end{block}
      \begin{itemize}
        \item If $\hat{A}_t>0$: increase action probability
        \item If $\hat{A}_t<0$: decrease action probability
        \item Problem: nothing stops very large policy changes
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Clipped PPO surrogate}
        \begin{minipage}[c][1.55cm][c]{\textwidth}
          \[
            L_t^{\mathrm{CLIP}}(\theta)=
            \min\!\left(
              r_t\hat{A}_t,
              \operatorname{clip}(r_t,1-\epsilon,1+\epsilon)\hat{A}_t
            \right)
          \]
        \end{minipage}
      \end{block}
      \begin{itemize}
        \item Allows updates near the old policy
        \item Removes incentive outside $[1-\epsilon,1+\epsilon]$
        \item $\epsilon$ controls how conservative the update is
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.2em}
  \centering
  \begin{tikzpicture}[
      scale=0.88,
      arrow/.style={-{Latex[length=1.8mm]}, thick, tuwblue!70},
      tick/.style={thick, tuwblue!70},

    \node[label, failred] at (6.0,0.28) {too far};
  \end{tikzpicture}
\end{frame}
